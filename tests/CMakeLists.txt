cmake_minimum_required(VERSION 3.22)
project(SlyceSoundQualityTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
get_filename_component(PATCH_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
add_executable(quality_math_test quality_math_test.cpp)
target_include_directories(quality_math_test PRIVATE "${PATCH_ROOT}/Source/DSP")
add_test(NAME PortableMath COMMAND quality_math_test)

option(SLYCE_BUILD_ENGINE_TESTS "Build DSP-engine smoke tests (requires JUCE or explicit local shim)" OFF)
option(SLYCE_TEST_LOCAL_SHIM "Use limited local compatibility shim, NOT the actual JUCE SDK" OFF)
set(JUCE_SOURCE_DIR "" CACHE PATH "Existing JUCE source checkout; no downloads are performed")
set(SLYCE_PROJECT_DIR "${PATCH_ROOT}" CACHE PATH "SLYCE project to test after patching; default tests bundled source")
if(SLYCE_BUILD_ENGINE_TESTS)
    # Byte-identical copies in a small staging tree retain the production
    # relative includes without needing missing UI/assets/build outputs.
    set(STAGE "${CMAKE_CURRENT_BINARY_DIR}/tested_source")
    file(MAKE_DIRECTORY "${STAGE}/AudioEngine" "${STAGE}/DSP")
    set(ENGINE_SOURCES "")
    foreach(ENGINE VoicePool SynthEngine SamplerEngine GranularEngine FXChain)
        foreach(EXT cpp h)
            configure_file("${SLYCE_PROJECT_DIR}/Source/AudioEngine/${ENGINE}.${EXT}"
                           "${STAGE}/AudioEngine/${ENGINE}.${EXT}" COPYONLY)
        endforeach()
        list(APPEND ENGINE_SOURCES "${STAGE}/AudioEngine/${ENGINE}.cpp")
    endforeach()
    configure_file("${SLYCE_PROJECT_DIR}/Source/DSP/SoundQuality.h" "${STAGE}/DSP/SoundQuality.h" COPYONLY)
    if(EXISTS "${SLYCE_PROJECT_DIR}/Source/DSP/Biquad.h")
        configure_file("${SLYCE_PROJECT_DIR}/Source/DSP/Biquad.h" "${STAGE}/DSP/Biquad.h" COPYONLY)
    else()
        # This is an unchanged header from the user's supplied base source.
        configure_file("support/Biquad.h" "${STAGE}/DSP/Biquad.h" COPYONLY)
    endif()
    add_executable(engine_smoke_test engine_smoke_test.cpp ${ENGINE_SOURCES})
    target_include_directories(engine_smoke_test PRIVATE "${STAGE}")
    if(SLYCE_TEST_LOCAL_SHIM)
        message(WARNING "LOCAL SHIM TEST: not a JUCE build. Reverb and decoder are placeholders.")
        target_include_directories(engine_smoke_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/local_shim")
    else()
        if(NOT EXISTS "${JUCE_SOURCE_DIR}/CMakeLists.txt")
            message(FATAL_ERROR "Set JUCE_SOURCE_DIR to a local JUCE checkout (or explicitly select limited local shim).")
        endif()
        add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/JUCE" EXCLUDE_FROM_ALL)
        target_link_libraries(engine_smoke_test PRIVATE juce::juce_dsp juce::juce_audio_formats)
        target_compile_definitions(engine_smoke_test PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    endif()
    add_test(NAME EngineSmoke COMMAND engine_smoke_test)
endif()

if(SLYCE_BUILD_ENGINE_TESTS)
    find_package(Threads REQUIRED)
    add_executable(looper_regression_test looper_regression_test.cpp)
    add_executable(sample_state_test sample_state_test.cpp)
    add_executable(factory_render_test factory_render_test.cpp "${SLYCE_PROJECT_DIR}/Source/AudioEngine/SynthEngine.cpp")
    foreach(target looper_regression_test sample_state_test factory_render_test)
        target_link_libraries(${target} PRIVATE Threads::Threads)
        if(SLYCE_TEST_LOCAL_SHIM)
            target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/local_shim")
        else()
            target_link_libraries(${target} PRIVATE juce::juce_dsp juce::juce_audio_formats)
            target_compile_definitions(${target} PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
        endif()
    endforeach()
    add_test(NAME LooperRegression COMMAND looper_regression_test)
    add_test(NAME SampleStateLimiter COMMAND sample_state_test)
    add_test(NAME FactoryReference COMMAND factory_render_test)
endif()
